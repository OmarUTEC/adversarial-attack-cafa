\end{block}

\begin{block}{Lavado de dinero}

de transacciones.

\section{Marco Teórico}

\begin{frame}{Fundamentos: Modelos Víctima y Escenarios de Ataque}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Arquitecturas Evaluadas (Clasificación):}
        \begin{itemize}
            \item \textbf{Regresión Logística (LR):} Lineal de referencia.
            \item \textbf{XGBoost:} Ensamble de árboles con gradient boosting optimizado.
            \item \textbf{Multi-Layer Perceptron (MLP):} Red neuronal densa clásica.
            \item \textbf{LSTM-Attention:} Red recurrente con mecanismo de atención aditiva.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Escenarios de Amenaza:}
        \begin{itemize}
            \item \textbf{Caja Blanca:} Acceso total a gradientes y parámetros internos. \textit{(CaFA)}
            \item \textbf{Caja Negra:} Solo entradas y salidas del modelo. \textit{(HopSkipJump, BoundaryAttack, SquareAttack)}
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Fundamentos: Formulación Matemática de la Evasión}
    La generación de un ejemplo adversarial se formula como un problema de optimización restringido:

    \[
    A_i(\theta, \mathcal{B}) = \max_{\delta \in \mathcal{B}} L\big(f_\theta(x_i + \delta), y_i\big)
    \]

    \vspace{0.2cm}
    El impacto global del ataque se cuantifica mediante el riesgo adversarial empírico:

    \[
    R_{P_n}(\theta, \mathcal{B}) = \frac{1}{n} \sum_{i=1}^{n} A_i(\theta, \mathcal{B})
    \]

    \textbf{Estrategias Algorítmicas bajo Evaluación:}
    \begin{itemize}
        \item \textbf{CaFA:} Ataque de caja blanca basado en gradientes sobre datos tabulares.
        \item \textbf{Square Attack:} Ataque de caja negra basado en consultas de tipo score.
        \item \textbf{HopSkipJump (HSJ) / Boundary Attack:} Ataques basados en la frontera de decisión geométrica.
    \end{itemize}
\end{frame}

\begin{frame}{Dataset usados}
    \textbf{Configuración de los Conjuntos de Datos:}
    \vspace{0.4cm}
    \begin{itemize}
        \item \textbf{Credit Card Fraud 2023:} \\

        Entorno de simulación masiva multiagente. Variables explícitas con alto significado semántico. Dataset con \textbf{imbalance extremo} (1:993), 5\,078\,208 transacciones, 15 features.
    \end{itemize}
\end{frame}

\begin{frame}{Protocolo de Evaluación Semántica: Mecanismo de Proyección}
    Para preservar la validez del ejemplo adversarial en el dominio transaccional, se introduce un operador de post-procesamiento $\Pi_{\mathcal{C}}$ que proyecta el vector perturbado al espacio de restricciones reales.

    \vspace{0.3cm}
    \textbf{Dimensiones de la Validación Semántica:}
    \begin{enumerate}
        \item \textbf{Inmutabilidad Estricta:} Bloqueo de atributos históricos inalterables (ID del cliente, marcas de tiempo fijas).
        \item \textbf{Consistencia de Tipos:} Operaciones de discretización y redondeo para conservar coherencia en variables numéricas discretas.
        \item \textbf{Límites Operativos:} Preservación de rangos lógicos del negocio ($\text{Monto} \ge 0$).
    \end{enumerate}
\end{frame}

% ================= SECCIÓN 5: CONCLUSIONES =================
\section{Conclusiones y Trabajo Futuro}

\begin{frame}{Conclusiones y Proyecciones}
    \begin{block}{Conclusiones Fundamentales}
        \begin{enumerate}
            \item \textbf{Ineficacia de métricas convencionales:} Clasificadores con rendimiento óptimo bajo condiciones normales colapsan ante perturbaciones intencionales sutiles.
            \item \textbf{El imbalance determina la vulnerabilidad:} En AMLworld, el desbalance extremo hace que 3 de 4 modelos sean inútiles para evaluación adversarial, independientemente de su arquitectura.
            \item \textbf{$L_0$ como métrica de sigilo:} El costo $L_0$ se establece como métrica indispensable para diseñar esquemas de detección de anomalías adversariales.
        \end{enumerate}
    \end{block}

    \vspace{0.2cm}
    \begin{exampleblock}{Trabajo Futuro}
        \begin{itemize}
            \item \textbf{Entrenamiento adversarial:} incorporar ejemplos adversariales al proceso de entrenamiento para aumentar la robustez activa de los modelos.
            \item \textbf{Defensa basada en $L_0$:} diseñar reglas de detección que alerten cuando múltiples features cambian simultáneamente.
            \item \textbf{Técnicas de balanceo:} evaluar SMOTE, submuestreo y pérdidas focales junto con robustez adversarial en datasets imbalanceados.
            \item \textbf{Transferibilidad:} analizar si ataques generados sobre un modelo son efectivos contra modelos desconocidos.
        \end{itemize}
    \end{exampleblock}
\end{frame}
